```python
def solution(genres, plays):
    # 장르별 노래 재생 횟수 합산
    # 해당 장르마다 노래 재생 횟수 카운트
    # 1<= 노래 수(N) <= 10000
    # 장르 종류(k) < 100

    # dictionary 에 장르를 키, [(i, 재생횟수)]를 값으로 하여 저장하기. O(N)
    # 배열에 저장할때 두번째 값을 기준으로 역방향으로 정렬되어 들어가도록.
    # 저장하면서 장르별로 재생 횟수 더하기
    play_sum_by_genre = {}
    plays_by_genre = {}
    for i in range(len(plays)):
        g = genres[i]
        p = plays[i]
        if g in plays_by_genre:
            # 삽입 정렬로 개선가능?
            plays_by_genre[g].append((i, p))
        else:
            plays_by_genre[g] = [(i, p)]
        if g in play_sum_by_genre:
            play_sum_by_genre[g] += p
        else:
            play_sum_by_genre[g] = p

    answer = []
    for genre, _ in sorted(play_sum_by_genre.items(), key=lambda item: -item[1]):
        plays = plays_by_genre[genre]
        counter = 0
        for i, p in sorted(plays, key=lambda play: (-play[1], play[0])):
            if counter == 2:
                break
            answer.append(i)
            counter += 1
    return answer
```

## 개선
defaultdict 사용으로 값 확인 안하고 넣기
zip으로 두 배열 한꺼번에 처리하기
정렬한뒤 슬라이싱으로 원하는 인덱스까지만 넣기
	list에 추가할때 extend로
```python
from collections import defaultdict

def solution(genres, plays):
    # 장르별 노래 재생 횟수 합산
    # 해당 장르마다 노래 재생 횟수 카운트
    # 1<= 노래 수(N) <= 10000
    # 장르 종류(k) < 100

    # dictionary 에 장르를 키, [(i, 재생횟수)]를 값으로 하여 저장하기. O(N)
    # 배열에 저장할때 두번째 값을 기준으로 역방향으로 정렬되어 들어가도록.
    # 저장하면서 장르별로 재생 횟수 더하기
    play_sum_by_genre = defaultdict(int)
    plays_by_genre = defaultdict(list)
    for i, (g, p) in enumerate(zip(genres, plays)):
        plays_by_genre[g].append((i, p))
        play_sum_by_genre[g] += p

    answer = []
    for genre, _ in sorted(play_sum_by_genre.items(), key=lambda item: -item[1]):
        sorted_plays = sorted(plays_by_genre[genre], key=lambda play: (-play[1], play[0]))
        answer.extend(i for i,_ in sorted_plays[:2])
    return answer
```
