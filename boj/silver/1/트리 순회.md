### 문제 이해
이진 트리 입력받아 전위 / 중위 / 후위 순회 구현
노드 개수 N
이후 노드와 왼 오 자식노드
A부터 차례대로 알파벳. 루트노드는 A
자식 노드 없으면 .

### 추상화
dfs로 그래프 탐색하면서 출력하기
트리 초기화?
### 계획
N번 반복하며 트리 초기화
별도 클래스를 만들자
	root, left, right을 필드로 하는 클래스 TreeNode
전위 순회
def traversal(node, mode):
	if mode == "preorder":
		print(node)
	if node.left != ".":
		traversal(node.left)
	if mode == "inorder":
		print(node)
	if node.right != ".":
		traversal(node.right)
	if mode == "postorder":
		print(node)
		
### 코드
```python
import sys
input = sys.stdin.readline


def solve():
    N = int(input())
    tree = {}
    for _ in range(N):
        root, left, right = input().split()
        tree[root] = (left, right)
  
    def preorder(node):
        if node == ".":
            return ""
        left, right = tree[node]
        return node + preorder(left) + preorder(right)
        
    def inorder(node):
        if node == ".":
            return ""
        left, right = tree[node]
        return inorder(left) + node + inorder(right)

    def postorder(node):
        if node == ".":
            return ""
        left, right = tree[node]
        return postorder(left) + postorder(right) + node
    
    print(preorder("A"))
    print(inorder("A"))
    print(postorder("A"))

if __name__ == "__main__":
    solve()
```
